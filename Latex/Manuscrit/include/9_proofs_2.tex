\section[Proof of Proposition 3.15]{Proof of \Cref{th:covariance_operator}}\label{sec:proof_Mercer_operator}

As $\Omega$ is compact and $\kappa$ continuous, $\kappa$ is bounded on $\Omega\times\Omega$, and $\mathcal{K}$ is compact.

Let $\phi,\phi'\in\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$. Cauchy-Schwarz inequality ensures that there exists a constant $c\in\mathbb{R}_+$ such that:
\begin{equation}
	\iint_{\Omega\times\Omega}{\left|\phi(y)^\top \kappa(x,y)^\top\phi'(x)\right|\:dx\:dy}\leq c\:\sqrt{\langle\phi|\phi\rangle}\:\sqrt{\langle\phi'|\phi'\rangle}<+\infty
\end{equation}

Fubini's integration theorem \cite{bib:Tonelli} can thus be used to give:
\begin{equation}
	\begin{aligned}
	\langle\mathcal{K}\phi|\phi'\rangle&=\int_{\Omega}{(\mathcal{K}\phi)(x)^\top\phi'(x)\:dx}\\
	&=\int_{\Omega}{\left(\int_{\Omega}{\kappa(x,y)\:\phi(y)\:dy}\right)^\top\phi'(x)\:dx}\\
	&=\int_{\Omega}{\int_{\Omega}{\phi(y)^\top \kappa(x,y)^\top\phi'(x)\:dy}\:dx}\\
	&=\int_{\Omega}{\int_{\Omega}{\phi(y)^\top \kappa(y,x)\:\phi'(x)\:dx}\:dy}\\
	&=\int_{\Omega}{\phi(y)^\top\left(\int_{\Omega}{\kappa(y,x)\:\phi'(x)\:dx}\right)dy}\\
	&=\int_{\Omega}{\phi(y)^\top(\mathcal{K}\phi')(y)\:dy}\\
	&=\langle\phi|\mathcal{K}\phi'\rangle
	\end{aligned}
\end{equation}

Let $\phi:\Omega\to\mathbb{R}^{d_z}$ be a continuous function, and let $n\in\mathbb{N}^*$. We define $\left(\Omega_i^{(n)}\right)_{i\in[\![1,n]\!]}$ a partition of $\Omega$ such that the diameter of each $\Omega_i^{(n)},\:i\in[\![1,n]\!]$ tends to $0$ when $n\to+\infty$. Here, the diameter of a subset of the Euclidean space $\mathbb{R}^{d_x}$ is defined as the maximum distance between two points of this subset. For $i\in[\![1,n]\!]$, we define $x_i^{(n)}\in\Omega_i^{(n)}$, and we let:
\begin{equation}
	a_i^{(n)}\triangleq\left(\int_{\Omega_i^{(n)}}{dx}\right)\phi\!\left(x_i^{(n)}\right)
\end{equation}

By continuity of $\phi$, the function:
\begin{equation}
	(x,y)\mapsto \phi(x)^\top \kappa(x,y)\:\phi(y)
\end{equation}
is continuous on the compact $\Omega\times\Omega$, so its Riemann sums converge:
\begin{equation}
	\sum_{i=1}^n\sum_{j=1}^n\left(a_i^{(n)}\right)^\top \kappa\left(x_i^{(n)},x_j^{(n)}\right)a_j^{(n)}\underset{n\to+\infty}{\longrightarrow}\iint_{\Omega\times\Omega}{\phi(x)^\top \kappa(x,y)\:\phi(y)\:dx\:dy}=\langle\phi|\mathcal{K}\phi\rangle
\end{equation}

The SPSDness of $\kappa$ ensures that each Riemann sum is non-negative. By passing to the limit, it follows that:
\begin{equation}
	\langle\phi|\mathcal{K}\phi\rangle\geq0\label{eq:fKf_geq0}
\end{equation}
for every continuous function $\phi$.

Finally, recall that the set of continuous functions of $\Omega$ is dense in $\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$. By continuity of:
\begin{equation}
	\phi\in\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})\mapsto\langle\phi|\mathcal{K}\phi\rangle
\end{equation}
the inequality \eqref{eq:fKf_geq0} extends to all $\phi\in\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$.

\section[Proof of Proposition 3.16]{Proof of \Cref{th:Mercer}}\label{sec:proof_Mercer}

By \Cref{th:covariance_operator}, $\mathcal{K}$ is a compact and self-adjoint operator on $\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$. The spectral theorem \cite{bib:spectral-theorem} thus states that there exists an orthonormal basis $(\psi_i)_{i\in\mathbb{N}^*}$ of $\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$ consisting of eigenfunctions of $\mathcal{K}$, with each corresponding eigenvalue $(\lambda_i)_{i\in\mathbb{N}^*}$ being real. Up to reordering, the sequence of eigenvalues can be considered to be non-increasing.

Since $\kappa$ is uniformly continuous on the compact $\Omega\times\Omega$, $\mathcal{K}\phi$ is continuous for every $\phi\in\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$. In particular, for each $i\in\mathbb{N}^*$ such that $\lambda_i\neq0$, the eigenfunction:
\begin{equation}
	\psi_i=\lambda_i^{-1}\mathcal{K}\psi_i
\end{equation}
is continuous. Without loss of generality, it must be noted that the $\psi_i$ associated with a null eigenvalue do not contribute to \eqref{eq:Mercer} and may thus be dismissed.

Since $(\psi_i)_{i\in\mathbb{N}^*}$ is an orthonormal basis of $\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$, the family $(\psi_i\:\psi_j^\top)_{i,j\in\mathbb{N}^*}$ is an orthonormal basis of $\mathcal{F}$. Let $\langle\cdot|\cdot\rangle_{\Omega\times\Omega}$ denote:
\begin{equation}
	(\phi,\phi')\in\mathcal{F}^2\mapsto\langle\phi|\phi'\rangle_{\Omega\times\Omega}\triangleq\iint_{\Omega\times\Omega}{\operatorname{tr}\left(\phi(x,y)^\top\phi'(x,y)\right)\:dx\:dy}
\end{equation}

Let $i,j\in\mathbb{N}^*$. We have:
\begin{equation}
	\begin{aligned}
		\langle \kappa|\psi_i\:\psi_j^\top\rangle_{\Omega\times\Omega}&=\iint_{\Omega\times\Omega}{\operatorname{tr}\left(\kappa(x,y)^\top\psi_i(x)\:\psi_j(y)^\top\right)dx\:dy}\\
		&=\iint_{\Omega\times\Omega}{\psi_j(y)^\top \kappa(x,y)^\top\psi_i(x)\:dx\:dy}\\
		&=\int_{\Omega}{\psi_j(y)^\top\left(\int_{\Omega}{\kappa(y,x)\:\psi_i(x)\:dx}\right)dy}\\
		&=\int_{\Omega}{\psi_j(y)^\top\mathcal{K}\psi_i(y)\:dy}\\
		&=\lambda_i\int_{\Omega}{\psi_j(y)^\top\:\psi_i(y)\:dy}\\
		&=\lambda_i\:\delta_{i,j}
	\end{aligned}
\end{equation}
where:
\begin{equation}
	\delta_{i,j}\triangleq\left\{\begin{aligned}
	1&\quad\text{if }i=j\\0&\quad\text{if }i\neq j
	\end{aligned}\right.
\end{equation}
is the Kronecker symbol.

Because $\kappa\in\mathcal{F}$, it follows from the completeness of $(\psi_i\:\psi_j^\top)_{i,j\in\mathbb{N}^*}$ that:
\begin{equation}
	\kappa_{r_{\mathrm{KL}}}(x,x')\triangleq\sum_{i=1}^{r_{\mathrm{KL}}}\lambda_i\:\psi_i(x)\:\psi_i(x')^\top
\end{equation}
converges to $\kappa$ in the $\mathcal{F}$ sense.

It must be noted that $\mathcal{L}^2$ convergence only identifies $\kappa$ up to a Lebesgue-null subset of $\Omega\times\Omega$, and gives no control at any specific point of the domain. In particular, it is not enough to justify substituting $\kappa_{r_{\mathrm{KL}}}$ for $\kappa$ in kriging scenarios. However, since $\kappa$ is continuous on $\Omega\times\Omega$, the convergence can be extended to absolute and uniform convergence on the entire domain \cite{bib:Karhunen-Loeve}. The argument applies Dini's theorem \cite{bib:Dini} to the non-decreasing sequence $\left(x\mapsto\operatorname{tr}\:\kappa_{r_{\mathrm{KL}}}(x,x)\right)_{r_{\mathrm{KL}}\in\mathbb{N}^*}$, whose limit is continuous on the compact $\Omega$, and extends the result off the diagonal through the Cauchy-Schwarz inequality.

Since $\mathcal{K}$ is SPSD, every eigenvalue is non-negative. Assume that there exists a rank $r_0\in\mathbb{N}^*$ such that $\lambda_i=0$ as soon as $i>r_0$. Then the decomposition \eqref{eq:Mercer} would reduce to the finite sum:
\begin{equation}
	\kappa=\sum_{i=1}^{r_0}\lambda_i\:\psi_i\:\psi_i^\top
\end{equation}
whose Gram matrices have rank at most $r_0$ and are thus singular at any $n$ distinct points with $nd_z>r_0$, contradicting the SPDness of $\kappa$ by \Cref{th:kernel_gram_equivalence}. Consequently:
\begin{equation}
	\lambda_1\geq\lambda_2\geq\cdots>0
\end{equation}

\section[Proof of Proposition 3.18]{Proof of \Cref{th:reduced_rank_kriging}}\label{sec:proof_reduced_rank_kriging}

As \eqref{eq:reduced_rank_Gram} is a rank $r_{\mathrm{KL}}$ perturbation of $\mathbf{A}\triangleq \mathbf{I}_{\tilde{N}}\otimes\tilde{\mathbf{R}}$, the Woodbury inversion formula (see \Cref{th:Woodbury}) can be used to reduce the $\tilde{N}d_z\times\tilde{N}d_z$ inversion of $\mathbf{G}(\tilde{\mathbf{x}})$ to an $r_{\mathrm{KL}}\times r_{\mathrm{KL}}$ one:
\begin{equation}
	\mathbf{G}(\tilde{\mathbf{x}})^{-1}\approx \mathbf{A}^{-1}-\mathbf{A}^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}
\end{equation}

Computing the centred SK estimator from \Cref{th:simple_kriging} using this approximation and the push-through identity (see \Cref{th:push_through}) yields:
\begin{equation}
	\begin{aligned}
	\hat{z}_{\mathrm{SK}}(x,\tilde{\mathbf{x}},\tilde{\mathbf{z}},0)&=\kappa(x,\tilde{\mathbf{x}})\:\mathbf{G}(\tilde{\mathbf{x}})^{-1}\tilde{\mathbf{z}}\\
	&\approx\boldsymbol{\Psi}(x)\:\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\left(\mathbf{I}-\mathbf{A}^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\right)\mathbf{A}^{-1}\tilde{\mathbf{z}}\\
	&=\boldsymbol{\Psi}(x)\:\boldsymbol{\Lambda}\left(\mathbf{I}-\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\right)\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\tilde{\mathbf{z}}
	\end{aligned}\label{eq:z_s_up_to_tilde_G}
\end{equation}

Recall that, by definition of $\tilde{\mathbf{G}}(\tilde{\mathbf{x}})$:
\begin{equation}
	\tilde{\mathbf{G}}(\tilde{\mathbf{x}})-\boldsymbol{\Lambda}^{-1}=\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})
\end{equation}
which gives:
\begin{equation}
	\begin{aligned}
	\mathbf{I}-\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}&=\mathbf{I}-\left(\tilde{\mathbf{G}}(\tilde{\mathbf{x}})-\boldsymbol{\Lambda}^{-1}\right)\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\\
	&=\mathbf{I}-\mathbf{I}+\boldsymbol{\Lambda}^{-1}\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\\
	&=\boldsymbol{\Lambda}^{-1}\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}
	\end{aligned}\label{eq:reduced_rank_lemma}
\end{equation}

Plugging this result into \eqref{eq:z_s_up_to_tilde_G} and using $\mathbf{A}^{-1}=\mathbf{I}_{\tilde{N}}\otimes\tilde{\mathbf{R}}^{-1}$ finally gives:
\begin{equation}
	\hat{z}_{\mathrm{SK}}(x,\tilde{\mathbf{x}},\tilde{\mathbf{z}},0)\approx\boldsymbol{\Psi}(x)\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\tilde{\mathbf{z}}
\end{equation}

For $\mathbf{R}_{\mathrm{SK}}$, reusing:
\begin{equation}
	\kappa(x,\tilde{\mathbf{x}})\:\mathbf{G}(\tilde{\mathbf{x}})^{-1}\approx\boldsymbol{\Psi}(x)\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}
\end{equation}
together with \eqref{eq:reduced_rank_lemma} yields:
\begin{equation}
	\begin{aligned}
	\mathbf{R}_{\mathrm{SK}}&=\kappa(x,x)+\mathbf{R}-\kappa(x,\tilde{\mathbf{x}})\:\mathbf{G}(\tilde{\mathbf{x}})^{-1}\kappa(\tilde{\mathbf{x}},x)\\
	&\approx\boldsymbol{\Psi}(x)\:\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(x)^\top+\mathbf{R}-\boldsymbol{\Psi}(x)\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top \mathbf{A}^{-1}\times\boldsymbol{\Psi}(\tilde{\mathbf{x}})\:\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(x)^\top\\
	&=\boldsymbol{\Psi}(x)\:\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(x)^\top+\mathbf{R}-\boldsymbol{\Psi}(x)\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\left(\tilde{\mathbf{G}}(\tilde{\mathbf{x}})-\boldsymbol{\Lambda}^{-1}\right)\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(x)^\top\\
	&=\boldsymbol{\Psi}(x)\:\boldsymbol{\Lambda}\:\boldsymbol{\Psi}(x)^\top+\mathbf{R}-\boldsymbol{\Psi}(x)\left(\boldsymbol{\Lambda}-\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\right)\boldsymbol{\Psi}(x)^\top\\
	&=\mathbf{R}+\boldsymbol{\Psi}(x)\:\tilde{\mathbf{G}}(\tilde{\mathbf{x}})^{-1}\boldsymbol{\Psi}(x)^\top
	\end{aligned}
\end{equation}

Finally, when $\tilde{\mathbf{R}}=\sigma^2\mathbf{I}_{d_z}$ with $\sigma>0$, substituting $\mathbf{A}^{-1}=\sigma^{-2}\mathbf{I}_{\tilde{N}d_z}$ yields the announced simplifications. For such $\sigma>0$, the matrix:
\begin{equation}
	\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\boldsymbol{\Psi}(\tilde{\mathbf{x}})+\sigma^2\boldsymbol{\Lambda}^{-1}\succ0
\end{equation}
is SPD, as the sum of the SPSD matrix $\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\boldsymbol{\Psi}(\tilde{\mathbf{x}})$ and the SPD matrix $\sigma^2\boldsymbol{\Lambda}^{-1}$. The map:
\begin{equation}
	\sigma\in\mathbb{R}_+^*\mapsto\left(\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\boldsymbol{\Psi}(\tilde{\mathbf{x}})+\sigma^2\boldsymbol{\Lambda}^{-1}\right)^{-1}
\end{equation}
is thus well-defined and continuous, and extends to $\sigma=0$ when $\boldsymbol{\Psi}(\tilde{\mathbf{x}})^\top\boldsymbol{\Psi}(\tilde{\mathbf{x}})\succ0$, \textit{i.e.} when $\boldsymbol{\Psi}(\tilde{\mathbf{x}})$ has full column rank. As $\hat{z}_{\mathrm{SK}}$ and $\mathbf{R}_{\mathrm{SK}}$ depend on $\sigma$ only through this inverse and an explicit factor $\sigma^2$, both expressions remain valid at $\sigma=0$ under this condition.

\section[Proof of Proposition 3.19]{Proof of \Cref{th:KL}}\label{sec:proof_KL}

Let $i\in\mathbb{N}^*$. Since $\psi_i$ is bounded on the compact $\Omega$, Tonelli's theorem \cite{bib:Tonelli} and the continuity of $\kappa$ give:
\begin{equation}
	\mathbb{E}\left[\int_{\Omega}{\|h(x)-m(x)\|^2\:dx}\right]=\int_{\Omega}{\operatorname{tr}\kappa(x,x)\:dx}<\infty
\end{equation}
so that $h-m\in\mathcal{L}^2(\Omega,\mathbb{R}^{d_z})$ a.s. Each $\langle h-m|\psi_i\rangle$ is thus well-defined and finite. This integrability also justifies the exchanges of expectation and integration below.

Since any finite linear combination of $(\xi_i)_{i\in\mathbb{N}^*}$ is itself a linear combination of the values of $h$, they are jointly Gaussian. Let $j\in\mathbb{N}^*$. It comes that:
\begin{equation}
	\mathbb{E}[\xi_i]=\frac{1}{\sqrt{\lambda_i}}\int_{\Omega}{\mathbb{E}[h(x)-m(x)]^\top\psi_i(x)\:dx}=0
\end{equation}
and:
\begin{equation}
	\begin{aligned}
	\mathbb{E}[\xi_i\:\xi_j]&=\frac{1}{\sqrt{\lambda_i\:\lambda_j}}\iint_{\Omega\times\Omega}{\psi_i(x)^\top\mathbb{E}[(h(x)-m(x))(h(x')-m(x'))^\top]\:\psi_j(x')\:dx\:dx'}\\
	&=\frac{1}{\sqrt{\lambda_i\:\lambda_j}}\iint_{\Omega\times\Omega}{\psi_i(x)^\top \kappa(x,x')\:\psi_j(x')\:dx\:dx'}\\
	&=\frac{1}{\sqrt{\lambda_i\:\lambda_j}}\int_{\Omega}{\psi_i(x)^\top\left(\int_{\Omega}{\kappa(x,x')\:\psi_j(x')\:dx'}\right)\:dx}\\
	&=\frac{\lambda_j}{\sqrt{\lambda_i\:\lambda_j}}\int_{\Omega}{\psi_i(x)^\top\psi_j(x)\:dx}\\
	&=\frac{\lambda_j}{\sqrt{\lambda_i\:\lambda_j}}\langle\psi_i|\psi_j\rangle\\
	&=\delta_{i,j}
	\end{aligned}
\end{equation}
using the fact that $\psi_i$ is an eigenfunction of the Mercer operator $\mathcal{K}$. Together with the jointly Gaussian nature of $(\xi_i)_{i\in\mathbb{N}^*}$, this proves that they are i.i.d. standard Gaussian variables.

Finally, for $r\in\mathbb{N}^*$, let:
\begin{equation}
	h_r:\left\{\begin{aligned}
	\Omega&\to\mathbb{R}^{d_z}\\
	x&\mapsto m(x)+\sum_{i=1}^r\sqrt{\lambda_i}\:\xi_i\:\psi_i(x)
	\end{aligned}\right.
\end{equation}

Fubini's theorem can be used to compute the following cross-covariance:
\begin{equation}
	\begin{aligned}
	\mathbb{E}\left[(h(x)-m(x))\:\xi_i\right]&=\frac{1}{\sqrt{\lambda_i}}\mathbb{E}[(h(x)-m(x))\langle h-m|\psi_i\rangle]\\
	&=\frac{1}{\sqrt{\lambda_i}}\mathbb{E}\left[(h(x)-m(x))\int_{\Omega}{(h(y)-m(y))^\top\psi_i(y)\:dy}\right]\\
	&=\frac{1}{\sqrt{\lambda_i}}\int_{\Omega}{\mathbb{E}\left[(h(x)-m(x))(h(y)-m(y))^\top\right]\psi_i(y)\:dy}\\
	&=\frac{1}{\sqrt{\lambda_i}}\int_{\Omega}{\kappa(x,y)\:\psi_i(y)\:dy}\\
	&=\sqrt{\lambda_i}\:\psi_i(x)
	\end{aligned}
\end{equation}
which can be used to obtain:
\begin{equation}
	\begin{aligned}
	\mathbb{E}\left[\|h(x)-h_r(x)\|^2\right]&=\mathbb{E}\left[\|h(x)-m(x)\|^2\right]-2\sum_{i=1}^r\sqrt{\lambda_i}\:\psi_i(x)^\top\mathbb{E}\left[(h(x)-m(x))\:\xi_i\right]\\
	&\phantom{{}={}}+\sum_{i=1}^r\sum_{j=1}^r\sqrt{\lambda_i\:\lambda_j}\:\mathbb{E}[\xi_i\:\xi_j]\:\psi_i(x)^\top\psi_j(x)\\
	&=\operatorname{tr}\kappa(x,x)-2\sum_{i=1}^r\lambda_i\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)+\sum_{i=1}^r\lambda_i\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)\hspace{-8pt}\\
	&=\operatorname{tr}\kappa(x,x)-\sum_{i=1}^r\lambda_i\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)\\
	&=\sum_{i=r+1}^{+\infty}\lambda_i\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)
	\end{aligned}
\end{equation}

By Mercer's theorem, this last sum is the tail of:
\begin{equation}
	\operatorname{tr}\kappa(x,x)=\sum_{i=1}^{+\infty}\lambda_i\:\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)
\end{equation}
and therefore converges to 0. The partial sums thus converge to $h$ in mean square, uniformly over $\Omega$.

\section[Proof of Proposition 3.20]{Proof of \Cref{th:KL_converse}}\label{sec:proof_KL_converse}

For $r\in\mathbb{N}^*$, let:
\begin{equation}
	h_r:\left\{\begin{aligned}
	\Omega&\to\mathbb{R}^{d_z}\\
	x&\mapsto m(x)+\sum_{i=1}^r\sqrt{\lambda_i}\:\xi_i\:\psi_i(x)
	\end{aligned}\right.
\end{equation}
be the partial sum of the series.

Let $r'\in\mathbb{N}^*$ such that $r'>r$ and $x\in\Omega$. Since the $(\xi_i)_{i\in\mathbb{N}^*}$ are i.i.d. standard Gaussian variables, it follows that:
\begin{equation}
	\begin{aligned}
	\mathbb{E}\left[\|h_r(x)-h_{r'}(x)\|^2\right]&=\sum_{i=r+1}^{r'}\lambda_i\:\|\psi_i(x)\|^2\\
	&=\sum_{i=r+1}^{r'}\lambda_i\:\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)
	\end{aligned}
\end{equation}

As for \Cref{th:KL}, this sum is the tail of the convergent series:
\begin{equation}
	\operatorname{tr}\kappa(x,x)=\sum_{i=1}^{+\infty}\lambda_i\:\operatorname{tr}\left(\psi_i(x)\:\psi_i(x)^\top\right)
\end{equation}
and tends to 0. The sequence $(h_r(x))_{r\in\mathbb{N}^*}$ is thus a Cauchy sequence in the space of square-integrable $\mathbb{R}^{d_z}$-valued random variables, which is complete. The sequence thus converges, and $h(x)$ is well-defined as its limit.

Let $n\in\mathbb{N}^*$, $x_1,\dots,x_n\in\Omega$ and $a_1,\dots,a_n\in\mathbb{R}^{d_z}$. For every $r\in\mathbb{N}^*$, the sum:
\begin{equation}
	\sum_{i=1}^na_i^\top h_r(x_i)
\end{equation}
is affine in $\xi_1,\dots,\xi_r$, and is thus Gaussian. Passing to the limit, this proves that:
\begin{equation}
	\sum_{i=1}^na_i^\top h(x_i)
\end{equation}
is also Gaussian, and so $h$ is a GP by \Cref{th:GP_definition}.

Since mean-square convergence carries the first two moments to the limit, the mean of $h$ is:
\begin{equation}
	\begin{aligned}
	\mathbb{E}[h(x)]&=\lim_{r\to+\infty}\mathbb{E}[h_r(x)]\\
	&=m(x)+\lim_{r\to+\infty}\sum_{i=1}^r\sqrt{\lambda_i}\:\mathbb{E}[\xi_i]\:\psi_i(x)\\
	&=m(x)
	\end{aligned}
\end{equation}
and its covariance is:
\begin{equation}
	\begin{aligned}
	\mathbb{E}\left[(h(x)-m(x))(h(x')-m(x'))^\top\right]&=\lim_{r\to+\infty}\sum_{i=1}^r\sum_{j=1}^r\sqrt{\lambda_i\:\lambda_j}\:\mathbb{E}[\xi_i\:\xi_j]\:\psi_i(x)\:\psi_j(x')^\top\\
	&=\lim_{r\to+\infty}\sum_{i=1}^r\sum_{j=1}^r\sqrt{\lambda_i\:\lambda_j}\:\delta_{i,j}\:\psi_i(x)\:\psi_j(x')^\top\\
	&=\sum_{i=1}^{+\infty}\lambda_i\:\psi_i(x)\:\psi_i(x')^\top\\
	&=\kappa(x,x')
	\end{aligned}
\end{equation}

\section[Proof of Proposition 3.22]{Proof of \Cref{th:native_RKHS}}\label{sec:proof_native_RKHS}

\textbf{Hilbert space.} Let:
\begin{equation}
	\mathcal{E}\triangleq\operatorname{span}\{\kappa(x,\cdot),\:x\in\Omega\}
\end{equation}
and let $\psi,\varphi\in\mathcal{E}$. There exist $n,n'\in\mathbb{N}^*$, $x_1,\dots,x_n,x'_1,\dots,x'_{n'}\in\Omega$ and $a_1,\dots,a_n,a'_1,\dots,a'_{n'}\in\mathbb{R}$ such that:
\begin{equation}
	\psi=\sum_{i=1}^na_i\:\kappa(x_i,\cdot)\quad\text{and}\quad\varphi=\sum_{j=1}^{n'}a'_j\:\kappa(x'_j,\cdot)
\end{equation}

Since $\kappa$ is symmetric, extending $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$ to $\mathcal{E}$ by bilinearity gives:
\begin{equation}
	\begin{aligned}
	\langle\psi\mid\varphi\rangle_{\mathcal{H}_\kappa}&=\sum_{i=1}^n\sum_{j=1}^{n'}a_i\:a'_j\:\kappa(x_i,x'_j)\\
	&=\sum_{j=1}^{n'}a'_j\:\psi(x'_j)\\
	&=\sum_{i=1}^na_i\:\varphi(x_i)
	\end{aligned}
\end{equation}

The second expression only depends on $\psi$ through its values, and the third one only depends on $\varphi$ through its values. The result thus does not depend on the decompositions of $\psi$ and $\varphi$, which are not unique in general, and $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$ is well-defined on $\mathcal{E}$. It is symmetric and bilinear by construction. Since $\kappa$ is SPSD, it is also positive semi-definite:
\begin{equation}
	\langle\psi\mid\psi\rangle_{\mathcal{H}_\kappa}=\sum_{i=1}^n\sum_{j=1}^na_i\:a_j\:\kappa(x_i,x_j)\geq0
\end{equation}

Let $x\in\Omega$. In particular, for $\varphi=\kappa(x,\cdot)$, it comes that:
\begin{equation}
	\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}=\psi(x)
\end{equation}

The Cauchy-Schwarz inequality, which holds for any symmetric positive semi-definite bilinear form, thus gives:
\begin{equation}
	\begin{aligned}
	\psi(x)^2&=\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}^2\\
	&\leq\langle\psi\mid\psi\rangle_{\mathcal{H}_\kappa}\:\langle \kappa(x,\cdot)\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}\\
	&=\langle\psi\mid\psi\rangle_{\mathcal{H}_\kappa}\:\kappa(x,x)
	\end{aligned}
\end{equation}

Consequently, $\langle\psi\mid\psi\rangle_{\mathcal{H}_\kappa}=0$ implies $\psi=0$, and $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$ is thus an inner product on $\mathcal{E}$.

The completion $\mathcal{H}_\kappa$ of $\mathcal{E}$ is a Hilbert space in which $\mathcal{E}$ is dense by construction, and its inner product is the extension of $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$ by continuity.

\textbf{Reproducing kernel.} We first identify the elements of $\mathcal{H}_\kappa$ with functions from $\Omega$ to $\mathbb{R}$. Let $x\in\Omega$ and $\psi\in\mathcal{H}_\kappa$, and let $(\psi_r)_{r\in\mathbb{N}^*}$ be a sequence of $\mathcal{E}$ that converges to $\psi$ in $\mathcal{H}_\kappa$. By continuity of the inner product, it comes that:
\begin{equation}
	\lim_{r\to+\infty}\psi_r(x)=\lim_{r\to+\infty}\langle\psi_r\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}=\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}
\end{equation}

The sequence $(\psi_r)_{r\in\mathbb{N}^*}$ thus converges pointwise to the function:
\begin{equation}
	x\in\Omega\mapsto\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}
\end{equation}
which only depends on $\psi$, and coincides with it when $\psi\in\mathcal{E}$. Assume that this function is null. Then $\psi$ is orthogonal to $\mathcal{E}$, and thus:
\begin{equation}
	\langle\psi\mid\psi\rangle_{\mathcal{H}_\kappa}=\lim_{r\to+\infty}\langle\psi\mid\psi_r\rangle_{\mathcal{H}_\kappa}=0
\end{equation}
so $\psi=0$. By linearity, two elements of $\mathcal{H}_\kappa$ associated with the same function are therefore equal.

Each element of $\mathcal{H}_\kappa$ can thus be identified with its associated function, and $\mathcal{H}_\kappa$ is a Hilbert space of functions from $\Omega$ to $\mathbb{R}$ which verifies:
\begin{equation}
	\forall x\in\Omega,\:\forall\psi\in\mathcal{H}_\kappa,\:\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}=\psi(x)
\end{equation}

Since $\kappa(x,\cdot)\in\mathcal{H}_\kappa$ for every $x\in\Omega$, \Cref{th:RKHS} is satisfied, and $\mathcal{H}_\kappa$ is an RKHS whose reproducing kernel is $\kappa$.

\textbf{Uniqueness.} Let $(\mathcal{H},\langle\cdot\mid\cdot\rangle_{\mathcal{H}})$ be an RKHS whose reproducing kernel is $\kappa$. By \Cref{th:RKHS}, it contains $\kappa(x,\cdot)$ for every $x\in\Omega$, and thus:
\begin{equation}
	\mathcal{E}\subseteq\mathcal{H}
\end{equation}

Let $x,x'\in\Omega$. The reproducing property of $\mathcal{H}$ gives:
\begin{equation}
	\langle \kappa(x,\cdot)\mid \kappa(x',\cdot)\rangle_{\mathcal{H}}=\kappa(x,x')=\langle \kappa(x,\cdot)\mid \kappa(x',\cdot)\rangle_{\mathcal{H}_\kappa}
\end{equation}
and the two inner products thus coincide on $\mathcal{E}$.

Let $\psi\in\mathcal{H}$ be orthogonal to $\mathcal{E}$. Then:
\begin{equation}
	\psi(x)=\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}}=0
\end{equation}
so $\psi=0$, and $\mathcal{E}$ is dense in $\mathcal{H}$.

Let $\psi\in\mathcal{H}$, and let $(\psi_r)_{r\in\mathbb{N}^*}$ be a sequence of $\mathcal{E}$ that converges to $\psi$ in $\mathcal{H}$. Since the two norms coincide on $\mathcal{E}$, $(\psi_r)_{r\in\mathbb{N}^*}$ is a Cauchy sequence of $\mathcal{E}$, which converges to some $\psi'\in\mathcal{H}_\kappa$. By continuity of the two inner products, it comes that:
\begin{equation}
	\begin{aligned}
	\psi(x)&=\langle\psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}}\\
	&=\lim_{r\to+\infty}\langle\psi_r\mid \kappa(x,\cdot)\rangle_{\mathcal{H}}\\
	&=\lim_{r\to+\infty}\psi_r(x)\\
	&=\lim_{r\to+\infty}\langle\psi_r\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}\\
	&=\langle\psi'\mid \kappa(x,\cdot)\rangle_{\mathcal{H}_\kappa}\\
	&=\psi'(x)
	\end{aligned}
\end{equation}
and:
\begin{equation}
	\|\psi\|_{\mathcal{H}}=\lim_{r\to+\infty}\|\psi_r\|_{\mathcal{H}}=\lim_{r\to+\infty}\|\psi_r\|_{\mathcal{H}_\kappa}=\|\psi'\|_{\mathcal{H}_\kappa}
\end{equation}

Therefore, $\psi=\psi'\in\mathcal{H}_\kappa$, and its norm is the same in both spaces. Since $\mathcal{H}$ is complete, exchanging the roles of $\mathcal{H}$ and $\mathcal{H}_\kappa$ gives the converse inclusion. Consequently, $\mathcal{H}=\mathcal{H}_\kappa$, and the two inner products are equal.

Finally, since $\tilde{\mathbf{z}}_{d|r}$ is unchanged when a trend is added to the observations, \Cref{th:explicit_residuals} can be applied to the observations $\tilde{\mathbf{z}}-g(\tilde{\mathbf{x}})\:\boldsymbol{\gamma}_0$:
\begin{equation}
	\begin{aligned}
	\tilde{\mathbf{z}}_{d|r}&=\left(\tilde{\mathbf{z}}^d-g(\tilde{\mathbf{x}}^d)\:\boldsymbol{\gamma}_0\right)-\kappa(\tilde{\mathbf{x}}^d,\tilde{\mathbf{x}}^r)\:\mathbf{G}(\tilde{\mathbf{x}}^r)^{-1}\left(\tilde{\mathbf{z}}^r-g(\tilde{\mathbf{x}}^r)\:\boldsymbol{\gamma}_0\right)\\
	&\qquad-\mathbf{U}(\tilde{\mathbf{x}}^d)\left(\hat{m}_{\mathrm{UK}}(\tilde{\mathbf{x}}^r,\tilde{\mathbf{z}}^r,g)-\boldsymbol{\gamma}_0\right)
	\end{aligned}
\end{equation}

The steps used for $\kappa^0_{d|r}(x)$ bound the norm of the first two terms by $\sqrt{\tilde{N}^d}\:\rho_r\:\tilde{z}_{\max}$. Moreover, \eqref{eq:entrywise_norms} gives:
\begin{equation}
	\begin{aligned}
	\left\|\hat{m}_{\mathrm{UK}}(\tilde{\mathbf{x}}^r,\tilde{\mathbf{z}}^r,g)-\boldsymbol{\gamma}_0\right\|_2&=\left\|\mathbf{M}(\tilde{\mathbf{x}}^r)\:g(\tilde{\mathbf{x}}^r)^\top \mathbf{G}(\tilde{\mathbf{x}}^r)^{-1}\left(\tilde{\mathbf{z}}^r-g(\tilde{\mathbf{x}}^r)\:\boldsymbol{\gamma}_0\right)\right\|_2\\
	&\leq\frac{\lambda_r^+}{\lambda_r^g}\:\frac{\sqrt{\tilde{N}^r}\:g_{\max}\:\sqrt{\tilde{N}^r}\:\tilde{z}_{\max}}{\lambda_r^-}\\
	&=\frac{\tilde{N}^rg_{\max}\:\lambda_r^+}{\lambda_r^g\:\lambda_r^-}\:\tilde{z}_{\max}
	\end{aligned}
\end{equation}
which concludes the proof by the triangle inequality.
