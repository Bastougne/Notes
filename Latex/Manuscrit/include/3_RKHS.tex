\section{Reproducing Kernel Hilbert Spaces}\label{sec:RKHS}

Beyond regression, the covariance kernel also induces a functional Hilbert space, known as a reproducing kernel Hilbert space (RKHS) \cite{bib:Aronszajn,bib:RKHS-Dumbgen}. The RKHS representation is complementary to the probabilistic viewpoint described throughout this chapter. Indeed, while the GP describes a probabilistic distribution over functions, the RKHS associated with its covariance kernel provides a deterministic functional space in which such functions can be represented, manipulated, or approximated. For the sake of simplicity, we assume that $d_z=1$, so that the reproducing property can be written in its most natural form.

\subsection{Reproducing Property and Definition}

Let $\Omega$ be an arbitrary set, and let $(\mathcal{H},\langle\cdot\mid\cdot\rangle_{\mathcal{H}})$ denote a Hilbert space of functions from $\Omega$ to $\mathbb{R}$. \Cref{th:RKHS,th:reproducing_kernel} describe the conditions for this space to be an RKHS.

\begin{definition}[Reproducing kernel Hilbert space]\label{th:RKHS}
	A Hilbert space $(\mathcal{H},\langle\cdot\mid\cdot\rangle_{\mathcal{H}})$ of functions from $\Omega$ to $\mathbb{R}$ is an RKHS if there exists a function $\kappa:\Omega\times\Omega\to\mathbb{R}$ such that:
	\begin{equation}
		\left\{\begin{aligned}
		\forall x\in\Omega,&\:\kappa(x,\cdot)\in\mathcal{H}\\
		\forall x\in\Omega,&\:\forall \psi\in\mathcal{H},\:\langle \psi\mid \kappa(x,\cdot)\rangle_{\mathcal{H}}=\psi(x)
		\end{aligned}\right.
	\end{equation}

	This second equation is called the reproducing property of $\mathcal{H}$.
\end{definition}

\begin{proposition}[Reproducing kernel]\label{th:reproducing_kernel}
	When such a $\kappa$ exists, it is symmetric and unique. It is called the reproducing kernel of $\mathcal{H}$.
\end{proposition}

\begin{proof}
	Let $\kappa_1$ satisfy \Cref{th:RKHS}, and let $x,x'\in\Omega$. Then:
	\begin{equation}
		\begin{aligned}
		\kappa_1(x,x')&=(\kappa_1(x,\cdot))(x')\\
		&=\langle \kappa_1(x,\cdot)\mid \kappa_1(x',\cdot)\rangle_{\mathcal{H}}\\
		&=\langle \kappa_1(x',\cdot)\mid \kappa_1(x,\cdot)\rangle_{\mathcal{H}}\\
		&=\kappa_1(x',x)
		\end{aligned}
	\end{equation}
	so $\kappa_1$ is symmetric.

	Assume that there exists another function $\kappa_2$ that verifies \Cref{th:RKHS}. It comes that:
	\begin{equation}
		\begin{aligned}
		\kappa_2(x,x')&=(\kappa_2(x,\cdot))(x')\\
		&=\langle \kappa_2(x,\cdot)\mid \kappa_1(x',\cdot)\rangle_{\mathcal{H}}\\
		&=\langle \kappa_1(x',\cdot)\mid \kappa_2(x,\cdot)\rangle_{\mathcal{H}}\\
		&=\kappa_1(x',x)\\
		&=\kappa_1(x,x')
		\end{aligned}
	\end{equation}
	and so $\kappa_2=\kappa_1$.
\end{proof}

It can be shown \cite{bib:RKHS-twelve-pages} that $\mathcal{H}$ is an RKHS if, and only if, the evaluation functional at every point of the domain:
\begin{equation}
	\operatorname{ev}_x:\psi\in\mathcal{H}\mapsto \psi(x)
\end{equation}
is continuous on $\mathcal{H}$ for every $x\in\Omega$. Its reproducing kernel is necessarily SPSD. Conversely, every SPSD kernel $\kappa$ admits a unique RKHS $\mathcal{H}_\kappa$ for which $\kappa$ is the reproducing kernel. Such an RKHS admits an explicit construction, as detailed in \Cref{th:native_RKHS}.

\begin{proposition}[Native RKHS]\label{th:native_RKHS}
	Let $\kappa$ be an SPSD kernel, let $\mathcal{H}_\kappa$ be the completion of the set spanned by $\{\kappa(x,\cdot),\:x\in\Omega\}$, and let $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$ be defined on the generators by:
	\begin{equation}
		\forall x,x'\in\Omega,\:\langle \kappa(x,\cdot)\mid \kappa(x',\cdot)\rangle_{\mathcal{H}_\kappa}\triangleq \kappa(x,x')
	\end{equation}

	Then $(\mathcal{H}_\kappa,\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa})$ is a Hilbert space and an RKHS. Its reproducing kernel is $\kappa$, and it is the only RKHS for which $\kappa$ is the reproducing kernel. It is called the native RKHS of $\kappa$.
\end{proposition}

\begin{proof}
	See \cref{sec:proof_native_RKHS}.
\end{proof}

The native RKHS is spanned by the functions $\kappa(x,\cdot)$, so every $x\in\Omega$ is naturally represented in $\mathcal{H}_\kappa$ by the $\kappa(x,\cdot)$ it generates. This correspondence defines the feature map, introduced by \Cref{th:feature_map}, and provides the representation on which the kernel methods of the following sections operate.

\begin{definition}[Feature map]\label{th:feature_map}
	Let $\kappa$ be an SPSD kernel on $\Omega$, and $(\mathcal{H}_\kappa,\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa})$ be its native RKHS. The feature map of $\mathcal{H}_\kappa$ is:
	\begin{equation}
		\Phi:\left\{\begin{aligned}
		\Omega&\to\mathcal{H}_\kappa\\
		x&\mapsto \kappa(x,\cdot)
		\end{aligned}\right.
	\end{equation}

	The reproducing property can thus be rewritten as:
	\begin{equation}
		\forall x\in\Omega,\:\forall \psi\in\mathcal{H}_\kappa,\:\psi(x)=\langle \psi\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}
	\end{equation}

	In particular, the kernel trick:
	\begin{equation}
		\forall x,x'\in\Omega,\:\langle\Phi(x)\mid\Phi(x')\rangle_{\mathcal{H}_\kappa}=\kappa(x,x')
	\end{equation}
	can be used to compute inner products in $\mathcal{H}_\kappa$ without explicitly constructing the feature map $\Phi$ or evaluating $\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa}$.
\end{definition}

\subsection{Link with Gaussian Processes}

In this section, we let $\kappa$ be an SPD kernel on $\Omega$, and $(\mathcal{H}_\kappa,\langle\cdot\mid\cdot\rangle_{\mathcal{H}_\kappa})$ be its native RKHS. \Cref{th:RKHS_regularisation} illustrates the relationship between the GP and RKHS viewpoints by showing that kriging using $\kappa$ is equivalent to solving a variational problem in $\mathcal{H}_\kappa$. It is a particular case of the representer theorem \cite{bib:representer,bib:representer-generalised}, presented by \Cref{th:representer}.

\begin{proposition}[Representer theorem]\label{th:representer}
	Let $\tilde{N}\in\mathbb{N}^*$, $\tilde{\mathbf{x}}\triangleq(\tilde{x}_1,\dots,\tilde{x}_{\tilde{N}})\in\Omega^{\tilde{N}}$, $E:\mathbb{R}^{\tilde{N}}\to\mathbb{R}$ and $G:\mathbb{R}_+\to\mathbb{R}$ strictly increasing.

	If the minimisation problem:
	\begin{equation}
		\underset{\psi\in\mathcal{H}_\kappa}{\operatorname{argmin}}(E(\psi(\tilde{x}_1),\dots,\psi(\tilde{x}_{\tilde{N}}))+G(\|\psi\|_{\mathcal{H}_\kappa}))\label{eq:RKHS_minimisation}
	\end{equation}
	admits a solution, then it admits a solution in the set spanned by the features of $\tilde{\mathbf{x}}$:
	\begin{equation}
		\mathcal{V}_{\tilde{\mathbf{x}}}\triangleq\operatorname{span}\{\Phi(\tilde{x}_1),\dots,\Phi(\tilde{x}_{\tilde{N}})\}\subseteq\mathcal{H}_\kappa
	\end{equation}

	In particular, this minimiser can be written as:
	\begin{equation}
		\psi_{\min}=\sum_{i=1}^{\tilde{N}}a_i\:\Phi(\tilde{x}_i)=a^\top\Phi(\tilde{\mathbf{x}})
	\end{equation}
	for some $a\triangleq[a_1,\dots,a_{\tilde{N}}]^\top\in\mathbb{R}^{\tilde{N}}$.
\end{proposition}

\begin{proof}
	For $\psi\in\mathcal{H}_\kappa$, we decompose it orthogonally as:
	\begin{equation}
		\exists\:\psi_{\mathcal{V}}\in\mathcal{V}_{\tilde{\mathbf{x}}},\:\psi_{\perp}\in\mathcal{V}_{\tilde{\mathbf{x}}}^\perp\mid\psi=\psi_{\mathcal{V}}+\psi_{\perp}
	\end{equation}
	where $\mathcal{V}_{\tilde{\mathbf{x}}}^\perp$ denotes the subspace of $\mathcal{H}_\kappa$ orthogonal to $\mathcal{V}_{\tilde{\mathbf{x}}}$.

	Let $i\in[\![1,\tilde{N}]\!]$. Since $\Phi(\tilde{x}_i)\in\mathcal{V}_{\tilde{\mathbf{x}}}$, the reproducing property gives
	\begin{equation}
		\psi_{\perp}(\tilde{x}_i)=\langle\psi_{\perp}\mid\Phi(\tilde{x}_i)\rangle_{\mathcal{H}_\kappa}=0
	\end{equation}
	and thus:
	\begin{equation}
		\psi(\tilde{x}_i)=\psi_{\mathcal{V}}(\tilde{x}_i)
	\end{equation}

	Moreover, by orthogonality:
	\begin{equation}
		\|\psi\|_{\mathcal{H}_\kappa}=\sqrt{\|\psi_{\mathcal{V}}\|_{\mathcal{H}_\kappa}^2+\|\psi_{\perp}\|_{\mathcal{H}_\kappa}^2}\geq\|\psi_{\mathcal{V}}\|_{\mathcal{H}_\kappa}
	\end{equation}
	with equality if and only if $\psi\in\mathcal{V}_{\tilde{\mathbf{x}}}$.

	Since $E$ depends only on the values of $\psi$ at
	$\tilde{x}_1,\dots,\tilde{x}_{\tilde{N}}$ and $G$ is strictly increasing, it comes that:
	\begin{equation}
		E(\psi(\tilde{x}_1),\dots,\psi(\tilde{x}_{\tilde{N}}))+G(\|\psi\|_{\mathcal{H}_\kappa})\geq E(\psi_{\mathcal{V}}(\tilde{x}_1),\dots,\psi_{\mathcal{V}}(\tilde{x}_{\tilde{N}}))+G(\|\psi_{\mathcal{V}}\|_{\mathcal{H}_\kappa})
	\end{equation}

	Therefore, any minimiser of \eqref{eq:RKHS_minimisation} must belong to $\mathcal{V}_{\tilde{\mathbf{x}}}$, provided that such a minimiser exists.
\end{proof}

\begin{proposition}[Regularised least-squares in an RKHS]\label{th:RKHS_regularisation}
	Let $\tilde{N}\in\mathbb{N}^*$, $\tilde{\mathbf{x}}\triangleq(\tilde{x}_1,\dots,\tilde{x}_{\tilde{N}})\in\Omega^{\tilde{N}}$ pairwise distinct, $\tilde{\mathbf{z}}\triangleq(\tilde{z}_1,\dots,\tilde{z}_{\tilde{N}})\in\mathbb{R}^{\tilde{N}}$ and $\alpha_{\mathrm{RKHS}}\geq0$. The regularised least-squares problem:
	\begin{equation}
		\underset{\psi\in\mathcal{H}_\kappa}{\operatorname{argmin}}\left(\frac{1}{\tilde{N}}\sum_{i=1}^{\tilde{N}}\left(\psi(\tilde{x}_i)-\tilde{z}_i\right)^2+\alpha_{\mathrm{RKHS}}\|\psi\|_{\mathcal{H}_\kappa}^2\right)\label{eq:RKHS_LS}
	\end{equation}
	admits a solution, unique for $\alpha_{\mathrm{RKHS}}>0$, and given by:
	\begin{equation}
		\hat{\psi}_{\mathrm{RKHS}}(x)=\kappa(x,\tilde{\mathbf{x}})\left(\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})+\alpha_{\mathrm{RKHS}}\tilde{N}\mathbf{I}_{\tilde{N}}\right)^{-1}\tilde{\mathbf{z}}
	\end{equation}
\end{proposition}

\begin{proof}
	Let $\psi\in\mathcal{H}_\kappa$, and let $\psi_{\mathcal{V}}$ denote its orthogonal projection onto $\mathcal{V}_{\tilde{\mathbf{x}}}$. As established in the proof of \Cref{th:representer}, replacing $\psi$ with $\psi_{\mathcal{V}}$ does not change the values $\psi(\tilde{x}_1),\dots,\psi(\tilde{x}_{\tilde{N}})$, and does not increase the norm. It is thus sufficient to minimise \eqref{eq:RKHS_LS} over $\mathcal{V}_{\tilde{\mathbf{x}}}$.

	Let $a\in\mathbb{R}^{\tilde{N}}$ such that $\psi=a^\top\Phi(\tilde{\mathbf{x}})$. By the reproducing property and the kernel trick, we have:
	\begin{equation}
		\left\{\begin{aligned}
		\psi(\tilde{\mathbf{x}})&=\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\:a\\
		\|\psi\|_{\mathcal{H}_\kappa}^2&=a^\top \kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\:a
		\end{aligned}\right.
	\end{equation}
	and the objective function of \eqref{eq:RKHS_LS} becomes:
	\begin{equation}
		J:a\in\mathbb{R}^{\tilde{N}}\mapsto\frac{1}{\tilde{N}}\left\|\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\:a-\tilde{\mathbf{z}}\right\|_2^2+\alpha_{\mathrm{RKHS}}\:a^\top \kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\:a
	\end{equation}

	Since $\kappa$ is SPD and the training points are pairwise distinct, $\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\succ0$. The function $J$ is thus strictly convex, and differentiating it with respect to $a$ yields:
	\begin{equation}
		\begin{aligned}
		&\frac{\partial J}{\partial a}=0\\
		\iff\quad&\frac{2}{\tilde{N}}\:\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\left(\left(\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})+\alpha_{\mathrm{RKHS}}\tilde{N}\mathbf{I}_{\tilde{N}}\right)a-\tilde{\mathbf{z}}\right)=0\\
		\iff\quad&a=\left(\kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})+\alpha_{\mathrm{RKHS}}\tilde{N}\mathbf{I}_{\tilde{N}}\right)^{-1}\tilde{\mathbf{z}}
		\end{aligned}
	\end{equation}
	which gives the expression of $\hat{\psi}_{\mathrm{RKHS}}$.

	Finally, when $\alpha_{\mathrm{RKHS}}>0$, the norm of $\psi$ is strictly greater than that of $\psi_{\mathcal{V}}$ for every $\psi\notin\mathcal{V}_{\tilde{\mathbf{x}}}$, and so $\hat{\psi}_{\mathrm{RKHS}}$ is the unique solution of \eqref{eq:RKHS_LS}. When $\alpha_{\mathrm{RKHS}}=0$, however, adding any element of $\mathcal{V}_{\tilde{\mathbf{x}}}^\perp$ to $\hat{\psi}_{\mathrm{RKHS}}$ leaves the objective function unchanged, and the solution is not unique in general.
\end{proof}

\begin{remark}[Simple kriging in the RKHS]
	For:
	\begin{equation}
		\alpha_{\mathrm{RKHS}}=\frac{1}{\tilde{N}}\:\tilde{\mathbf{R}}
	\end{equation}
	the least-squares estimator $\hat{\psi}_{\mathrm{RKHS}}$ coincides with the centred SK estimator of \Cref{th:simple_kriging}. Simple kriging can therefore be interpreted as a regularised least-squares regression in $\mathcal{H}_\kappa$ \cite{bib:GP-kernel-methods}. In the noiseless case $\tilde{\mathbf{R}}=0$, the SK estimator is also the solution of the minimum-norm interpolation problem:
	\begin{equation}
		\underset{\psi\in\mathcal{H}_\kappa}{\operatorname{argmin}}\left\{\|\psi\|_{\mathcal{H}_\kappa}\:\middle|\:\forall i\in[\![1,\tilde{N}]\!],\:\psi(\tilde{x}_i)=\tilde{z}_i\right\}
	\end{equation}
\end{remark}

\begin{remark}[GP and RKHS correspondence]
	The GP and RKHS viewpoints thus give two interpretations of the same estimator. The first one is probabilistic, as the mean of the conditional distribution of \Cref{th:simple_kriging_conditional}, while the second one is deterministic, as the solution of the variational problem \eqref{eq:RKHS_LS}. Moreover, under the assumptions of \Cref{th:Mercer}, it can be shown \cite{bib:Rasmussen-relationships} that $\mathcal{H}_\kappa$ is the set of the functions:
	\begin{equation}
		\varphi=\sum_{i=1}^{+\infty}a_i\:\psi_i
	\end{equation}
	such that:
	\begin{equation}
		\|\varphi\|_{\mathcal{H}_\kappa}^2=\sum_{i=1}^{+\infty}\frac{a_i^2}{\lambda_i}<+\infty
	\end{equation}
	where $(\lambda_i,\psi_i)_{i\in\mathbb{N}^*}$ are the Mercer eigenpairs of $\kappa$. Since the eigenvalues converge to $0$, the coefficients $a_i$ must decay fast enough for this norm to be finite, which imposes a regularity condition on the elements of $\mathcal{H}_\kappa$.

	Choosing a kernel thus amounts to choosing a regularity class for the unknown field. In particular, for $\Omega=\mathbb{R}^{d_x}$, the native RKHS of a Matérn kernel of smoothness $\nu$ is the Sobolev space of order $\nu+d_x/2$, up to an equivalent norm \cite{bib:Wendland}, whereas that of the SE kernel only contains infinitely differentiable functions.
\end{remark}

\Cref{th:power_function} extends this correspondence to the estimation error, by showing that the error committed on a function of unit norm of $\mathcal{H}_\kappa$ is at most the standard deviation of the same error for a GP of covariance kernel $\kappa$.

\begin{proposition}[Worst-case error interpretation]\label{th:power_function}
	Let $\tilde{N}\in\mathbb{N}^*$, $\tilde{\mathbf{x}}\in\Omega^{\tilde{N}}$ pairwise distinct, $x\in\Omega$ and $\mathbf{W}\in\mathbb{R}^{\tilde{N}}$. Let $\psi\in\mathcal{H}_\kappa$ and $h\sim\mathcal{GP}(0,\kappa)$. Then:
	\begin{equation}
		\left|\mathbf{W}^\top\psi(\tilde{\mathbf{x}})-\psi(x)\right|\leq\|\psi\|_{\mathcal{H}_\kappa}\:\sqrt{\mathbb{V}\left[\mathbf{W}^\top h(\tilde{\mathbf{x}})-h(x)\right]}\label{eq:worst_case_error}
	\end{equation}
	with equality when:
	\begin{equation}
		\psi=\mathbf{W}^\top\Phi(\tilde{\mathbf{x}})-\Phi(x)
	\end{equation}

	Furthermore, the right-hand side of \eqref{eq:worst_case_error} is minimised by the weights of the SK estimator of \Cref{th:simple_kriging} for $\mathbf{R}=0$ and $\tilde{\mathbf{R}}=0$, which gives:
	\begin{equation}
		\left|\hat{z}_{\mathrm{SK}}(x,\tilde{\mathbf{x}},\psi(\tilde{\mathbf{x}}),0)-\psi(x)\right|\leq\|\psi\|_{\mathcal{H}_\kappa}\:\sqrt{\mathbf{R}_{\mathrm{SK}}(x,\tilde{\mathbf{x}})}
	\end{equation}
	where $\sqrt{\mathbf{R}_{\mathrm{SK}}(x,\tilde{\mathbf{x}})}$ is known as the power function \cite{bib:Wendland}.
\end{proposition}

\begin{proof}
	Let $w_1,\dots,w_{\tilde{N}}$ denote the components of $\mathbf{W}$, and let:
	\begin{equation}
		\varphi\triangleq \mathbf{W}^\top\Phi(\tilde{\mathbf{x}})-\Phi(x)=\sum_{i=1}^{\tilde{N}}w_i\:\Phi(\tilde{x}_i)-\Phi(x)
	\end{equation}

	The reproducing property and the Cauchy-Schwarz inequality give:
	\begin{equation}
		\begin{aligned}
		\left|\mathbf{W}^\top\psi(\tilde{\mathbf{x}})-\psi(x)\right|&=\left|\sum_{i=1}^{\tilde{N}}w_i\:\langle\psi\mid\Phi(\tilde{x}_i)\rangle_{\mathcal{H}_\kappa}-\langle\psi\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}\right|\\
		&=\left|\langle\psi\mid\varphi\rangle_{\mathcal{H}_\kappa}\right|\\
		&\leq\|\psi\|_{\mathcal{H}_\kappa}\:\|\varphi\|_{\mathcal{H}_\kappa}
		\end{aligned}
	\end{equation}
	with equality when $\psi=\varphi$.

	We also have:
	\begin{equation}
		\begin{aligned}
		\|\varphi\|_{\mathcal{H}_\kappa}^2&=\sum_{i=1}^{\tilde{N}}\sum_{j=1}^{\tilde{N}}w_i\:w_j\:\langle\Phi(\tilde{x}_i)\mid\Phi(\tilde{x}_j)\rangle_{\mathcal{H}_\kappa}-2\sum_{i=1}^{\tilde{N}}w_i\:\langle\Phi(\tilde{x}_i)\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}\\
		&\qquad+\langle\Phi(x)\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}\\
		&=\sum_{i=1}^{\tilde{N}}\sum_{j=1}^{\tilde{N}}w_i\:w_j\:\kappa(\tilde{x}_i,\tilde{x}_j)-2\sum_{i=1}^{\tilde{N}}w_i\:\kappa(\tilde{x}_i,x)+\kappa(x,x)\\
		&=\mathbf{W}^\top \kappa(\tilde{\mathbf{x}},\tilde{\mathbf{x}})\:\mathbf{W}-2\mathbf{W}^\top \kappa(\tilde{\mathbf{x}},x)+\kappa(x,x)
		\end{aligned}
	\end{equation}
	which is exactly \eqref{eq:generic_kriging_variance} for $\mathbf{R}=0$ and $\tilde{\mathbf{R}}=0$, and thus the variance of $\mathbf{W}^\top h(\tilde{\mathbf{x}})-h(x)$. As established in the proof of \Cref{th:simple_kriging}, this variance is minimised by the SK weights, for which it is equal to $\mathbf{R}_{\mathrm{SK}}(x,\tilde{\mathbf{x}})$.
\end{proof}

It must be noted that this interpretation only holds for noiseless observations of a function of $\mathcal{H}_\kappa$, whose norm is unknown in practice. Furthermore, it can be shown \cite{bib:GP-kernel-methods} that the sample paths of a GP of covariance kernel $\kappa$ almost surely do not belong to $\mathcal{H}_\kappa$ when it is infinite-dimensional.

\subsection{Kernel Mean Embedding}

Beyond evaluating functions inside $\mathcal{H}_\kappa$, the feature map can also be used to represent the moments of a probability distribution. \Cref{th:KME} introduces the kernel mean embedding (KME) of a random variable \cite{bib:KME}, while \Cref{th:joint_KME} generalises this concept to two random variables.

\begin{definition}[Kernel mean embedding]\label{th:KME}
	Let $X$ be a random variable on $\Omega$ with probability density $p$, and $\varphi\in\mathcal{H}_\kappa$. The KME of $X$ is:
	\begin{equation}
		\mu_X\triangleq\mathbb{E}_p\left[\Phi(X)\right]=\int_{\Omega}{\Phi(x)\:p(x)\:dx}\in\mathcal{H}_\kappa\label{eq:KME}
	\end{equation}

	By the reproducing property, the expectation of $\varphi(X)$ can then be expressed as an inner product in $\mathcal{H}_\kappa$:
	\begin{equation}
		\begin{aligned}
		\mathbb{E}_p\left[\varphi(X)\right]&=\int_{\Omega}{\varphi(x)\:p(x)\:dx}\\
		&=\int_{\Omega}{\langle\varphi\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}\:p(x)\:dx}\\
		&=\left\langle\varphi\:\middle|\:\int_{\Omega}{\Phi(x)\:p(x)\:dx}\right\rangle_{\mathcal{H}_\kappa}\\
		&=\langle\varphi\mid\mu_X\rangle_{\mathcal{H}_\kappa}
		\end{aligned}\label{eq:KME_expectation}
	\end{equation}
\end{definition}

Since \eqref{eq:KME} integrates a function taking its values in $\mathcal{H}_\kappa$, its existence is not granted. \Cref{th:KME_existence} gives a sufficient condition \cite{bib:RKHS-Dumbgen}, which every bounded kernel satisfies.

\begin{proposition}[Existence of the KME]\label{th:KME_existence}
	If $\mathbb{E}_p\left[\sqrt{\kappa(X,X)}\right]<\infty$, then $\mu_X$ is well defined, as the unique element of $\mathcal{H}_\kappa$ for which \eqref{eq:KME_expectation} holds for all $\varphi\in\mathcal{H}_\kappa$. Moreover:
	\begin{equation}
		\forall x\in\Omega,\:\mu_X(x)=\mathbb{E}_p\left[\kappa(X,x)\right]
	\end{equation}
\end{proposition}

\begin{proof}
	Let $\varphi\in\mathcal{H}_\kappa$. We have:
	\begin{equation}
		\begin{aligned}
		\left|\mathbb{E}_p\left[\varphi(X)\right]\right|&\leq\mathbb{E}_p\left[\left|\varphi(X)\right|\right]\\
		&=\mathbb{E}_p\left[\left|\langle\varphi\mid\Phi(X)\rangle_{\mathcal{H}_\kappa}\right|\right]\\
		&\leq\|\varphi\|_{\mathcal{H}_\kappa}\:\mathbb{E}_p\left[\|\Phi(X)\|_{\mathcal{H}_\kappa}\right]\\
		&=\|\varphi\|_{\mathcal{H}_\kappa}\:\mathbb{E}_p\left[\sqrt{\kappa(X,X)}\right]
		\end{aligned}
	\end{equation}

	By linearity of the expectation, $\varphi\mapsto\mathbb{E}_p\left[\varphi(X)\right]$ is thus a continuous linear form on $\mathcal{H}_\kappa$. By the Riesz representation theorem \cite{bib:Riesz}, it is the inner product with a unique element of $\mathcal{H}_\kappa$. This element is the meaning given to the integral of \eqref{eq:KME}, and it satisfies \eqref{eq:KME_expectation} by construction.

	Finally, let $x\in\Omega$. The reproducing property and \eqref{eq:KME_expectation} applied to $\varphi=\Phi(x)$ give:
	\begin{equation}
		\mu_X(x)=\langle\mu_X\mid\Phi(x)\rangle_{\mathcal{H}_\kappa}=\mathbb{E}_p\left[(\Phi(x))(X)\right]=\mathbb{E}_p\left[\kappa(X,x)\right]
	\end{equation}
\end{proof}

It must be noted that two different probability densities may have the same KME. A kernel for which the map $p\mapsto\mu_X$ is injective is called a characteristic kernel \cite{bib:characteristic-kernels}. In this case, $\mu_X$ uniquely determines $p$. In particular, the SE and Matérn kernels of \cref{sec:covariance_kernels} are characteristic.

\begin{definition}[Joint KME]\label{th:joint_KME}
	Let $X$ and $Z$ be random variables on two sets $\Omega_X$ and $\Omega_Z$, with joint probability density $p_{XZ}$. Let $\kappa_X$ and $\kappa_Z$ be SPD kernels on $\Omega_X$ and $\Omega_Z$, with respective native RKHSs $(\mathcal{H}_{\kappa_X},\langle\cdot\mid\cdot\rangle_{\mathcal{H}_{\kappa_X}})$ and $(\mathcal{H}_{\kappa_Z},\langle\cdot\mid\cdot\rangle_{\mathcal{H}_{\kappa_Z}})$, and feature maps $\Phi_X$ and $\Phi_Z$. The joint KME of $(X,Z)$ is defined as:
	\begin{equation}
		\begin{aligned}
		\mu_{XZ}&\triangleq\mathbb{E}_{p_{XZ}}\left[\Phi_X(X)\otimes\Phi_Z(Z)\right]\\
		&=\iint_{\Omega_X\times\Omega_Z}{\Phi_X(x)\otimes\Phi_Z(z)\:p_{XZ}(x,z)\:dx\:dz}\\
		&\in\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}
		\end{aligned}
	\end{equation}
	where $\otimes$ denotes the tensor product in the product space.
\end{definition}

The space $\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}$ is the completion of the set spanned by the tensors $\varphi\otimes\psi$, for $\varphi\in\mathcal{H}_{\kappa_X}$ and $\psi\in\mathcal{H}_{\kappa_Z}$, under the inner product defined on these generators by:
\begin{equation}
	\langle\varphi\otimes\psi\mid\varphi'\otimes\psi'\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}\triangleq\langle\varphi\mid\varphi'\rangle_{\mathcal{H}_{\kappa_X}}\langle\psi\mid\psi'\rangle_{\mathcal{H}_{\kappa_Z}}
\end{equation}

Each of its elements $T$ also acts as a continuous linear operator from $\mathcal{H}_{\kappa_Z}$ to $\mathcal{H}_{\kappa_X}$, the tensor $\varphi'\otimes\psi'$ mapping $\psi$ to $\langle\psi'\mid\psi\rangle_{\mathcal{H}_{\kappa_Z}}\:\varphi'$, so that:
\begin{equation}
	\forall\varphi\in\mathcal{H}_{\kappa_X},\:\forall\psi\in\mathcal{H}_{\kappa_Z},\:\langle\varphi\mid T\psi\rangle_{\mathcal{H}_{\kappa_X}}=\langle\varphi\otimes\psi\mid T\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}
\end{equation}

Similarly to \Cref{th:KME_existence}, \Cref{th:cross_covariance} gives a sufficient condition for the existence of $\mu_{XZ}$. It also generalises \eqref{eq:KME_expectation} to expectations involving both $X$ and $Z$ \cite{bib:kernel-embeddings}.

\begin{proposition}[Existence of the joint KME]\label{th:cross_covariance}
	With the notations of \Cref{th:joint_KME}, if $\mathbb{E}_{p_{XZ}}\left[\sqrt{\kappa_X(X,X)\:\kappa_Z(Z,Z)}\right]<\infty$, then $\mu_{XZ}$ is well defined and:
	\begin{equation}
		\forall\varphi\in\mathcal{H}_{\kappa_X},\:\forall\psi\in\mathcal{H}_{\kappa_Z},\:\langle\varphi\mid\mu_{XZ}\:\psi\rangle_{\mathcal{H}_{\kappa_X}}=\mathbb{E}_{p_{XZ}}\left[\varphi(X)\:\psi(Z)\right]
	\end{equation}
\end{proposition}

\begin{proof}
	Let $x\in\Omega_X$ and $z\in\Omega_Z$. We have:
	\begin{equation}
		\|\Phi_X(x)\otimes\Phi_Z(z)\|_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}^2=\kappa_X(x,x)\:\kappa_Z(z,z)
	\end{equation}

	The same arguments presented in the proof of \Cref{th:KME_existence} can be applied in $\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}$ to define $\mu_{XZ}$ as the unique element of this space such that, for all $T\in\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}$:
	\begin{equation}
		\langle T\mid\mu_{XZ}\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}=\mathbb{E}_{p_{XZ}}\left[\langle T\mid\Phi_X(X)\otimes\Phi_Z(Z)\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}\right]
	\end{equation}

	Let $\varphi\in\mathcal{H}_{\kappa_X}$ and $\psi\in\mathcal{H}_{\kappa_Z}$. Applying the previous results with $T=\varphi\otimes\psi$ yields:
	\begin{equation}
		\begin{aligned}
		\langle\varphi\mid\mu_{XZ}\:\psi\rangle_{\mathcal{H}_{\kappa_X}}&=\langle\varphi\otimes\psi\mid\mu_{XZ}\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}\\
		&=\mathbb{E}_{p_{XZ}}\left[\langle\varphi\otimes\psi\mid\Phi_X(X)\otimes\Phi_Z(Z)\rangle_{\mathcal{H}_{\kappa_X}\otimes\mathcal{H}_{\kappa_Z}}\right]\\
		&=\mathbb{E}_{p_{XZ}}\left[\langle\varphi\mid\Phi_X(X)\rangle_{\mathcal{H}_{\kappa_X}}\langle\psi\mid\Phi_Z(Z)\rangle_{\mathcal{H}_{\kappa_Z}}\right]\\
		&=\mathbb{E}_{p_{XZ}}\left[\varphi(X)\:\psi(Z)\right]
		\end{aligned}
	\end{equation}
\end{proof}

\Cref{th:conditional_KME} introduces the embedding of a conditional law, and \Cref{th:conditional_operator} shows that it depends linearly on the feature of the value taken by the conditioning variable \cite{bib:conditional-embeddings}. In the following, $\mu_{ZX}$ and $\mu_{XX}$ denote the joint KMEs of $(Z,X)$ and of $(X,X)$, the latter being taken under the kernel $\kappa_X$ alone.

\begin{definition}[Conditional KME]\label{th:conditional_KME}
	With the notations of \Cref{th:joint_KME}, let $p_{Z|X}$ be the conditional probability density of $Z$ given $X$, and $x\in\Omega_X$. The conditional KME of $Z$ given $X=x$ is:
	\begin{equation}
		\mu_{Z|x}\triangleq\mathbb{E}\left[\Phi_Z(Z)|X=x\right]=\int_{\Omega_Z}{\Phi_Z(z)\:p_{Z|X}(z|x)\:dz}\in\mathcal{H}_{\kappa_Z}\label{eq:conditional_KME}
	\end{equation}

	As in \Cref{th:KME}, it can be used to express conditional expectations in terms of inner products:
	\begin{equation}
		\forall\psi\in\mathcal{H}_{\kappa_Z},\:\mathbb{E}[\psi(Z)|X=x]=\langle\psi\mid\mu_{Z|x}\rangle_{\mathcal{H}_{\kappa_Z}}
	\end{equation}
\end{definition}

\begin{proposition}[Conditional embedding operator]\label{th:conditional_operator}
	With the notations of \Cref{th:conditional_KME}, assume that:
	\begin{equation}
		\psi\mapsto\mathbb{E}[\psi(Z)|X=\cdot]
	\end{equation}
	is a continuous linear operator from $\mathcal{H}_{\kappa_Z}$ to $\mathcal{H}_{\kappa_X}$.

	We also let $\mu_{Z|X}$ be its adjoint, \textit{i.e.} the continuous linear operator from $\mathcal{H}_{\kappa_X}$ to $\mathcal{H}_{\kappa_Z}$ such that:
	\begin{equation}
		\forall\varphi\in\mathcal{H}_{\kappa_X},\:\forall\psi\in\mathcal{H}_{\kappa_Z},\:\langle\psi\mid\mu_{Z|X}\:\varphi\rangle_{\mathcal{H}_{\kappa_Z}}=\langle\mathbb{E}[\psi(Z)|X=\cdot]\mid\varphi\rangle_{\mathcal{H}_{\kappa_X}}
	\end{equation}

	The conditional KME is then linear in the feature of the conditioning value:
	\begin{equation}
		\forall x\in\Omega_X,\:\mu_{Z|x}=\mu_{Z|X}\Phi_X(x)
	\end{equation}
	and $\mu_{Z|X}$ maps the embeddings of $X$ onto those of $(Z,X)$ and $Z$:
	\begin{equation}
		\left\{\begin{aligned}
		\mu_{Z|X}\:\mu_{XX}&=\mu_{ZX}\\
		\mu_{Z|X}\:\mu_X&=\mu_Z
		\end{aligned}\right.
	\end{equation}
\end{proposition}

\begin{proof}
	Let $\varphi\in\mathcal{H}_{\kappa_X}$, $\psi\in\mathcal{H}_{\kappa_Z}$ and $x\in\Omega_X$. We have:
	\begin{equation}
		\begin{aligned}
		\langle\psi\mid\mu_{Z|x}\rangle_{\mathcal{H}_{\kappa_Z}}&=\mathbb{E}[\psi(Z)|X=x]\\
		&=\langle\mathbb{E}[\psi(Z)|X=\cdot]\mid\Phi_X(x)\rangle_{\mathcal{H}_{\kappa_X}}\\
		&=\langle\psi\mid\mu_{Z|X}\Phi_X(x)\rangle_{\mathcal{H}_{\kappa_Z}}
		\end{aligned}
	\end{equation}

	Similarly, the tower property of conditional expectation yields:
	\begin{equation}
		\begin{aligned}
		\langle\psi\mid\mu_{Z|X}\:\mu_{XX}\:\varphi\rangle_{\mathcal{H}_{\kappa_Z}}&=\langle\mathbb{E}[\psi(Z)|X=\cdot]\mid\mu_{XX}\:\varphi\rangle_{\mathcal{H}_{\kappa_X}}\\
		&=\mathbb{E}\left[\mathbb{E}[\psi(Z)|X]\:\varphi(X)\right]\\
		&=\mathbb{E}_{p_{XZ}}\left[\psi(Z)\:\varphi(X)\right]\\
		&=\langle\psi\mid\mu_{ZX}\:\varphi\rangle_{\mathcal{H}_{\kappa_Z}}
		\end{aligned}
	\end{equation}
	while the law of total expectation yields:
	\begin{equation}
		\begin{aligned}
		\langle\psi\mid\mu_{Z|X}\:\mu_X\rangle_{\mathcal{H}_{\kappa_Z}}&=\langle\mathbb{E}[\psi(Z)|X=\cdot]\mid\mu_X\rangle_{\mathcal{H}_{\kappa_X}}\\
		&=\mathbb{E}\left[\mathbb{E}[\psi(Z)|X]\right]\\
		&=\mathbb{E}\left[\psi(Z)\right]\\
		&=\langle\psi\mid\mu_Z\rangle_{\mathcal{H}_{\kappa_Z}}
		\end{aligned}
	\end{equation}
\end{proof}
